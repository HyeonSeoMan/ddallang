// 측정용 정적 렌더링(저장소 밖). 시드 상태로 방 화면을 그려 HTML로 저장한다.
import { act } from "@testing-library/react";
import { writeFileSync } from "node:fs";
import { createStore } from "jotai";
import type { ReactNode } from "react";
import { it, vi } from "vitest";
import { renderApp } from "@/test/render";
import * as fixtures from "@/test/fixtures";
import { INVITE_CODE, seedRoom } from "@/test/room-seed";
import { roomBasePath } from "@/features/room/room-header";
import { RecordPage, RoomPage, SettlementPage, StatusPage } from "@/features/room/room-pages";
import { RoomShell } from "@/features/room/room-shell";
import { applySnapshotAtom, setConnectionAtom, startEpochAtom } from "@/features/room/state";

const OUT = "/private/tmp/claude-501/-Users-hsjeong-orca-workspaces-ddallang-design/83dce767-75b4-4806-b124-b2feb56a142e/scratchpad/v2render/out";
const nav = vi.hoisted(() => ({ pathname: null as string | null }));
vi.mock("next/navigation", () => ({ usePathname: () => nav.pathname, useRouter: () => ({ replace: () => {}, push: () => {} }) }));
const BASE = roomBasePath(INVITE_CODE);

function save(name: string, path: string, page: ReactNode, setup: (store: ReturnType<typeof createStore>) => void) {
  nav.pathname = BASE + path;
  const store = createStore();
  setup(store);
  const { container, unmount } = renderApp(<RoomShell inviteCode={INVITE_CODE}>{page}</RoomShell>, store);
  writeFileSync(`${OUT}/${name}.html`, container.innerHTML);
  unmount();
}

const UNIT = "가나다라마바사아자차카타";
const NAMES = ["Abcdefghijkl", "Mnopqrstuvwx", "가나다라마바사아자차카타"];
function maxSnapshot(store: ReturnType<typeof createStore>) {
  seedRoom(store);
  const epoch = store.set(startEpochAtom);
  const participants = fixtures.participants.map((p, i) => ({ ...p, nickname: NAMES[i]! }));
  store.set(applySnapshotAtom, {
    epoch,
    room: { ...fixtures.room, name: "아주아주긴방이름입니다열두자", unit: UNIT, status: "IN_PROGRESS" },
    me: fixtures.hostMember,
    participants,
    balances: [
      { participantId: "p-1", nickname: NAMES[0]!, status: "ACTIVE", order: 1, balance: 999_999_999_999 },
      { participantId: "p-3", nickname: NAMES[2]!, status: "INACTIVE", order: 3, balance: 0 },
      { participantId: "p-2", nickname: NAMES[1]!, status: "ACTIVE", order: 2, balance: -999_999_999_999 },
    ],
    transactions: [
      { ...fixtures.transactions[0]!, amount: 999_999_999_999, reason: "가".repeat(50), round: "123456789012" },
      { ...fixtures.transactions[1]!, amount: 999_999_999_999 },
    ],
    nextCursor: "c",
    transactionFilter: null,
    settlement: { kind: "EXPECTED", items: [{ payerId: "p-2", payerNickname: NAMES[1]!, payeeId: "p-1", payeeNickname: NAMES[0]!, amount: 999_999_999_999 }] },
  });
}

it("render", () => {
  save("status", "", <StatusPage />, (s) => seedRoom(s));
  save("record", "/record", <RecordPage />, (s) => seedRoom(s));
  save("settlement", "/settlement", <SettlementPage />, (s) => seedRoom(s));
  save("room", "/room", <RoomPage />, (s) => seedRoom(s));
  save("ended", "/settlement", <SettlementPage />, (s) => seedRoom(s, { status: "ENDED" }));
  save("offline", "", <StatusPage />, (s) => { seedRoom(s); act(() => s.set(setConnectionAtom, "disconnected")); });
  save("max-status", "", <StatusPage />, maxSnapshot);
  save("max-settlement", "/settlement", <SettlementPage />, maxSnapshot);
  save("max-room", "/room", <RoomPage />, maxSnapshot);
});
